75 void Config::LoadFromConfigDir(const std::string& path, const bool force)
76 {
77 try
78 {
79 const auto dir_path = canonical(std::filesystem::path(path));
80 if (!is_directory(dir_path))
81 {
82 LOG_ERROR(g_logger)
83 << "Config::LoadFromConfigDir - Invalid configuration directory provided. "
84 << "Directory path: " << dir_path.string();
85 return;
86 }
87
88 std::vector<FileMeta> file_metas;
89 for (const auto& entry : std::filesystem::recursive_directory_iterator(
90 dir_path,
91 std::filesystem::directory_options::skip_permission_denied
92 ))
93 {
94 try
95 {
96 if (entry.is_regular_file() &&
97 entry.path().extension() == ".yml")
98 {
99 const auto ftime = entry.last_write_time().time_since_epoch().count();
100 file_metas.push_back({
101 entry.path().string(),
102 static_cast<uint64_t>(ftime),
103 entry.status().type()
104 });
105 }
106 }
107 catch (std::filesystem::filesystem_error& e)
108 {
109 LOG_WARN(g_logger)
110 << "Config::LoadFromConfigDir - Skipping inaccessible file. "
111 << "File path: " << e.path1().string();
112 }
113 }
114
115 // 并行处理文件
116 std::for_each(std::execution::par, file_metas.begin(), file_metas.end(),
117 [&](const FileMeta& meta)
118 {
119 {
120 std::shared_lock read_lock(s_rwmutex);
121 if (!force && s_file2ModifyTime.count(meta.path) &&
122 s_file2ModifyTime.at(meta.path) == meta.mtime)
123 {
124 return;
125 }
126 }
127 try
128 {
129 const YAML::Node root = YAML::LoadFile(meta.path);
130 std::unique_lock write_lock(s_rwmutex);
131 LoadFromYaml(root);
132 s_file2ModifyTime[meta.path] = meta.mtime;
133 LOG_INFO(g_logger)
134 << "Config::LoadFromConfigDir - Successfully loaded configuration file. "
135 << "File path: " << meta.path;
136 }
137 catch (const YAML::Exception& e)
138 {
139 LOG_ERROR(g_logger)
140 << "Config::LoadFromConfigDir - YAML Error encountered while processing configuration file. "
141 << "File path: " << meta.path
142 << " | Error details: " << e.what();
143 }
144 catch (const std::exception& e)
145 {
146 LOG_ERROR(g_logger)
147 << "Config::LoadFromConfigDir - Failed to load configuration file. "
148 << "File path: " << meta.path
149 << " | Error details: " << e.what();
150 }
151 }
152 );
153 }

191 void ListAllMember(const std::string& prefix, const YAML::Node& node,
192 std::list<std::pair<std::string, const YAML::Node>>& output)
193 {
194 struct StackItem
195 {
196 std::string prefix;
197 YAML::Node node;
198
199 StackItem(std::string p, const YAML::Node& n)
200 : prefix(std::move(p)), node(n)
201 {
202 }
203 };
204 std::stack<StackItem> stack;
205 stack.emplace(prefix, node);
206
207 while (!stack.empty())
208 {
209 auto current = std::move(stack.top());
210 stack.pop();
211 if (current.prefix.find_first_not_of("abcdefghijklmnopqrstuvwxyz._012345678")
212 != std::string::npos)
213 {
214 LOG_ERROR(g_logger)
215 << "ListAllMember - Invalid configuration name encountered. "
216 << "Invalid name: " << prefix;
217 continue;
218 }
219
220 output.emplace_back(current.prefix, current.node);
221 if (current.node.IsMap())
222 {
224 std::vector<std::pair<YAML::Node, YAML::Node>> children;
225
227 for (auto it = current.node.begin(); it != current.node.end(); ++it)
228 {
229 children.emplace_back(it->first, it->second);
230 }
231
233 for (auto rit = children.rbegin(); rit != children.rend(); ++rit)
234 {
235 std::string new_prefix = current.prefix.empty()
236 ? rit->first.Scalar()
237 : current.prefix + "." + rit->first.Scalar();
238
239 stack.emplace(std::move(new_prefix), rit->second);
240 }
241 }
242 }
243 }

248 struct LogIniter

250 LogIniter()
251 {
252 global_log_defines->addListener([](const std::set<LogDefine>& old_value,
253 const std::set<LogDefine>& new_value)-> void
254 {
255 LOG_INFO(g_logger) << "LogIniter - The log configuration has been updated.";
256 for (auto& i : new_value)
257 {
258 auto it = old_value.find(i);
259 std::shared_ptr<Logger> logger = nullptr;
260 if (it == old_value.end())
261 {
262 logger = LOG_NAME(i.name);
263 }
264 else
265 {
266 if (!(i == *it))
267 {
268 logger = LOG_NAME(i.name);
269 }
270 else
271 {
272 continue;
273 }
274 }
275 logger->setLevel(i.level);
276 if (!i.formatter.empty())
277 {
278 logger->setFormatter(i.formatter);
279 }
280
281 logger->clearAppenders();
282 for (const auto& [type, level, formatter, file] : i.appenders)
283 {
284 std::shared_ptr<LogAppender> ap = nullptr;
285 if (type == 1)
286 {
287 ap = std::make_shared<FileLogAppender>(file);
288 }
289 else if (type == 2)
290 {
291 ap = std::make_shared<StdoutLogAppender>();
292 }
293 else
294 {
295 continue;
296 }
297 ap->setLevel(level);
298 if (!formatter.empty())
299 {
300 if (auto fmt = std::make_shared<LogFormatter>(formatter); !fmt->isError())
301 {
302 ap->setFormatter(fmt);
303 }
304 else
305 {
306 LOG_WARN(g_logger) << "LogIniter - Invalid appender configuration. "
307 << "Log name: " << i.name
308 << " | Appender type: " << type
309 << " | Formatter: " << formatter
310 << " is invalid." << std::endl;
311 }
312 }
313 logger->addAppender(ap);
314 }
315 }
316
317 for (auto& i : old_value)
318 {
319 if (auto it = new_value.find(i); it == new_value.end())
320 {
321 const auto logger = LOG_NAME(i.name);
322 logger->setLevel(static_cast<LogLevel::Level>(0));
323 logger->clearAppenders();
324 }
325 }
326 });
