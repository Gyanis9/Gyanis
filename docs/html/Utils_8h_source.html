58 template <class ValueType, class MapType, class KeyType>
59 ValueType GetParamValue(const MapType& param_map, const KeyType& key, const ValueType& default_value = ValueType())
60 {
61 // 查找键值对是否存在
62 auto iterator = param_map.find(key);
63
64 // 如果找不到指定的键，返回默认值
65 if (iterator == param_map.end())
66 {
67 return default_value;
68 }
69
70 try
71 {
72 // 使用 boost::lexical_cast 将找到的值转换为期望的类型
73 return boost::lexical_cast<ValueType>(iterator->second);
74 }
75 catch (...)
76 {
77 // 捕获所有异常，若转换失败则返回默认值
78 }
79
80 // 转换失败时返回默认值
81 return default_value;
82 }
