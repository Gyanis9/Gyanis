65 void Scheduler::run()
66 {
67 set_hook_enable(true);
68 setThis();
69 const auto& idle_fiber = std::make_shared<Fiber>([this] { this->idle(); });
70 std::shared_ptr<Fiber> cb_fiber = nullptr;
71 FiberAndThread ft;
72 while (true)
73 {
74 ft.reset();
75 bool tickle_me = false;
76 bool is_active = false;
77 {
78 std::lock_guard lock(m_mutex);
79 auto it = m_fibers.begin();
80 while (it != m_fibers.end())
81 {
82 if (it->thread != -1 && it->thread != base::GetThreadID())
83 {
84 ++it;
85 tickle_me = true;
86 continue;
87 }
88
89 ASSERT(it->fiber || it->cb);
90 if (it->fiber && it->fiber->getState() == Fiber::EXEC)
91 {
92 ++it;
93 continue;
94 }
95
96 ft = *it;
97 m_fibers.erase(it++);
98 ++m_activeThreadCount;
99 is_active = true;
100 break;
101 }
102 tickle_me |= it != m_fibers.end();
103 }
104 if (tickle_me)
105 {
106 tickle();
107 }
108 if (ft.fiber && ft.fiber->getState() != Fiber::TERM && ft.fiber->getState() != Fiber::EXCEPT)
109 {
110 ft.fiber->resume();
111 --m_activeThreadCount;
112 if (ft.fiber->getState() == Fiber::READY)
113 {
114 schedule(ft.fiber);
115 }
116 else if (ft.fiber->getState() != Fiber::TERM && ft.fiber->getState() != Fiber::EXCEPT)
117 {
118 ft.fiber->setHold();
119 }
120 ft.reset();
121 }
122 else if (ft.cb)
123 {
124 if (cb_fiber)
125 {
126 cb_fiber->reset(ft.cb);
127 }
128 else
129 {
130 cb_fiber = std::make_shared<Fiber>(ft.cb);
131 }
132 ft.reset();
133 cb_fiber->resume();
134 --m_activeThreadCount;
135 if (cb_fiber->getState() == Fiber::READY)
136 {
137 schedule(cb_fiber);
138 cb_fiber.reset();
139 }
140 else if (cb_fiber->getState() == Fiber::EXCEPT || cb_fiber->getState() == Fiber::TERM)
141 {
142 cb_fiber->reset(nullptr);
143 }
144 else
145 {
146 cb_fiber->setHold();
147 cb_fiber.reset();
148 }
149 }
150 else
151 {
152 if (is_active)
153 {
154 --m_activeThreadCount;
155 continue;
156 }
157 if (idle_fiber->getState() == Fiber::TERM)
158 {
159 LOG_DEBUG(g_logger)
160 << "Scheduler::run - Idle fiber execution completed. "
161 << " | Status: Normal";
162 break;
163 }
164 ++m_idleThreadCount;
165 idle_fiber->resume();
166 --m_idleThreadCount;
167 if (idle_fiber->getState() != Fiber::TERM
168 && idle_fiber->getState() != Fiber::EXCEPT)
169 {
170 idle_fiber->setHold();
171 }
172 }
173 }
