138 void Env::printHelp()
139 {

141
142 // 定义ANSI颜色代码
143 constexpr auto RESET = "\033[0m";
144
145 try
146 {
147 constexpr auto YELLOW = "\033[1;33m";
148 constexpr auto GREEN = "\033[1;32m";
149 constexpr auto CYAN = "\033[1;36m";
150 // 获取终端宽度
151 winsize w{};
152 size_t console_width = 80;
153 if (ioctl(STDOUT_FILENO, TIOCGWINSZ, &w) == 0)
154 {
155 console_width = w.ws_col;
156 }
157
158 std::cout << "\n" << CYAN << m_program << " - Usage" << RESET << "\n\n";
159
160 std::cout << "Usage: " << YELLOW << m_program << RESET << " [options]\n\n";
161
162 // 选项表头
163 std::cout << GREEN << std::left << std::setw(20) << "OPTION"
164 << "DESCRIPTION" << RESET << "\n";
165 std::wstring_convert<std::codecvt_utf8<wchar_t>> converter;
166 std::cout << converter.to_bytes(std::wstring(console_width, L'\u2500')) << "\n";
167
168 // 打印选项详情
169 for (const auto& [option, desc] : m_helps)
170 {
171 constexpr auto MAGENTA = "\033[1;35m";
172 constexpr size_t OPTION_COL_WIDTH = 20;
173 // 选项列（带颜色）
174 std::cout << MAGENTA << std::left << std::setw(OPTION_COL_WIDTH)
175 << ("-" + option) << RESET;
176
177 // 智能换行描述
178 size_t current_pos = OPTION_COL_WIDTH;
179 std::istringstream iss(desc);
180 std::string word;
181
182 while (iss >> word)
183 {
184 if (const size_t space_left = console_width - current_pos; word.length() > space_left)
185 {
186 std::cout << "\n" << std::string(OPTION_COL_WIDTH, ' ');
187 current_pos = OPTION_COL_WIDTH;
188 }
189 std::cout << word << " ";
190 current_pos += word.length() + 1;
191 }
192 std::cout << "\n";
193 }
194 std::cout << std::endl;
195 }
196 catch (...)
197 {
198 std::cout << RESET; // 确保异常时重置颜色
199 throw;
200 }
