280 std::shared_ptr<protocol::Message> RockMessageDecoder::parseForm(const std::shared_ptr<stream::Stream>& stream)
281 {
282 try
283 {
284 RockMsgHeader header;
285 if (stream->readFixSize(&header, sizeof(header)) <= 0)
286 {
287 LOG_ERROR(g_logger)
288 << "RockMessageDecoder header decoding failed. "
289 << "An error occurred while decoding the message header.";
290 return nullptr;
291 }
292
293 if (memcmp(header.magic, s_rock_magic, sizeof(s_rock_magic)) != 0)
294 {
295 LOG_ERROR(g_logger)
296 << "RockMessageDecoder header magic validation failed. "
297 << "The magic value in the header is incorrect or invalid.";
298 return nullptr;
299 }
300
301 if (header.version != 0x1)
302 {
303 LOG_ERROR(g_logger)
304 << "RockMessageDecoder header version mismatch. "
305 << "Expected version: 0x1, but found: " << header.version;
306 return nullptr;
307 }
308
309 header.length = base::byteswapOnLittleEndian(header.length);
310 if (static_cast<uint32_t>(header.length) >= g_rock_protocol_max_length->getValue())
311 {
312 LOG_ERROR(g_logger)
313 << "RockMessageDecoder header length exceeds maximum allowed. "
314 << "Header length: " << header.length
315 << " | Maximum allowed length: " << g_rock_protocol_max_length->getValue();
316 return nullptr;
317 }
318 auto ba = std::make_shared<stream::ByteArray>();
319 if (stream->readFixSize(ba, header.length) <= 0)
320 {
321 LOG_ERROR(g_logger)
322 << "RockMessageDecoder failed to read body. "
323 << "Body length: " << header.length
324 << " | An error occurred while reading the message body.";
325 return nullptr;
326 }
327
328 ba->setPosition(0);
329 if (header.flag & 0x1)
330 {
331 const auto zstream = stream::ZlibStream::CreateGzip(false);
332 if (zstream->write(ba, -1) != Z_OK)
333 {
334 LOG_ERROR(g_logger)
335 << "RockMessageDecoder GZIP decompression failed. "
336 << "An error occurred while decompressing the data using GZIP.";
337 return nullptr;
338 }
339 if (zstream->flush() != Z_OK)
340 {
341 LOG_ERROR(g_logger)
342 << "RockMessageDecoder GZIP flush failed. "
343 << "An error occurred while flushing the decompressed data in GZIP format.";
344 return nullptr;
345 }
346 ba = zstream->getByteArray();
347 }
348 const uint8_t type = ba->readFuint8();
349 std::shared_ptr<protocol::Message> msg;
350 switch (type)
351 {
352 case protocol::Message::REQUEST:
353 msg = std::make_shared<RockRequest>();
354 break;
355 case protocol::Message::RESPONSE:

357 break;

359 msg = std::make_shared<RockNotify>();
360 break;
361 default:
362 LOG_ERROR(g_logger)

364 << "Type value: " << static_cast<int>(type)
365 << ". Please verify the message type.";
366 return nullptr;
367 }
368
369 if (!msg->parseFromByteArray(ba))
370 {
371 LOG_ERROR(g_logger)
372 << "RockMessageDecoder deserialization failed. "
373 << "Invalid type: " << static_cast<int>(type)
374 << ". An error occurred while parsing the byte array.";
