61 std::shared_ptr<HttpRequest> WSSession::handleShake()
62 {
63 std::shared_ptr<HttpRequest> request = nullptr;
64 do
65 {
66 request = recvRequest();
67 if (!request)
68 {
69 LOG_INFO(g_logger)
70 << "Invalid HTTP request received. "
71 << "Please verify the request format and try again.";
72 break;
73 }
74 if (strcasecmp(request->getHeader("Upgrade").c_str(), "websocket") != 0)
75 {
76 LOG_INFO(g_logger)
77 << "HTTP header 'Upgrade' is not set to 'websocket. "
78 << "The connection upgrade request is invalid.";
79 break;
80 }
81 if (strcasecmp(request->getHeader("Connection").c_str(), "Upgrade") != 0)
82 {
83 LOG_INFO(g_logger)
84 << "HTTP header 'Connection' is not set to 'Upgrade'. "
85 << "The connection header is invalid for a WebSocket handshake.";
86 break;
87 }
88 if (request->getHeaderAs<int>("Sec-WebSocket-Version") != 13)
89 {
90 LOG_INFO(g_logger)
91 << "HTTP header 'Sec-WebSocket-Version' is not set to '13'. "
92 << "The WebSocket version is invalid for the handshake.";
93 break;
94 }
95 const std::string key = request->getHeader("Sec-WebSocket-Key");
96 if (key.empty())
97 {
98 LOG_INFO(g_logger)
99 << "HTTP header 'Sec-WebSocket-Key' is missing or null. "
100 << "The WebSocket key is required for a valid handshake.";
101 break;
102 }
103
104 std::string value = key + "258EAFA5-E914-47DA-95CA-C5AB0DC85B11";
105 value = base::base64encode(base::sha1sum(value));
106 request->setWebSocket(true);
107
108 const auto response = request->createResponse();
109 response->setStatus(HttpStatus::SWITCHING_PROTOCOLS);
110 response->setWebsocket(true);
111 response->setReason("Web Socket Protocol Handshake");
112 response->setHeader("Upgrade", "websocket");
113 response->setHeader("Connection", "Upgrade");
114 response->setHeader("Sec-WebSocket-Accept", value);
115
116 sendResponse(response);
117 LOG_DEBUG(g_logger) << *request;
118 LOG_DEBUG(g_logger) << *response;
119 return request;
120 }
121 while (false);
122 if (request)
123 {
124 LOG_INFO(g_logger) << *request;
125 }
126 return nullptr;
127 }

164 std::shared_ptr<WSFrameMessage> WSRecvMessage(stream::Stream* stream, const bool client)
165 {
166 int opcode = 0;
167 std::string data;
168 int cur_len = 0;
169 do
170 {
171 WSFrameHead ws_head{};
172 if (stream->readFixSize(&ws_head, sizeof(ws_head)) <= 0)
173 {
174 break;
175 }
176 LOG_DEBUG(g_logger)
177 << "WebSocket Frame Header: "
178 << ws_head.toString();
179
180 if (ws_head.opcode == WSFrameHead::PING)
181 {
182 LOG_INFO(g_logger) << "PING";
183 if (WSPong(stream) <= 0)
184 {
185 break;
186 }
187 }
188 else if (ws_head.opcode == WSFrameHead::PONG)
189 {
190 }
191 else if (ws_head.opcode == WSFrameHead::CONTINUE
192 || ws_head.opcode == WSFrameHead::TEXT_FRAME
193 || ws_head.opcode == WSFrameHead::BIN_FRAME)
194 {
195 if (!client && !ws_head.mask)
196 {
197 LOG_INFO(g_logger)
198 << "WebSocket Frame Header 'mask' is not set to 1. "
199 << "The mask field should be set to 1 for a valid WebSocket frame.";
200 break;
201 }
202 uint64_t length = 0;
203 if (ws_head.payload == 126)
204 {
205 uint16_t len = 0;
206 if (stream->readFixSize(&len, sizeof(len)) <= 0)
207 {
208 break;
209 }
210 length = base::byteswapOnLittleEndian(len);
211 }
212 else if (ws_head.payload == 127)
213 {
214 uint64_t len = 0;
215 if (stream->readFixSize(&len, sizeof(len)) <= 0)
216 {
217 break;
218 }
219 length = base::byteswapOnLittleEndian(len);
220 }
221 else
222 {
223 length = ws_head.payload;
224 }
225
226 if (cur_len + length >= g_websocket_message_max_size->getValue())
227 {
228 LOG_WARN(g_logger)
229 << "WebSocket Frame Message length exceeds maximum allowed. "
230 << "Max allowed length: " << g_websocket_message_max_size->getValue()
231 << " | Current length: " << (cur_len + length);
232
233 break;
234 }
235
236 char mask[4] = {};
237 if (ws_head.mask)
238 {
239 if (stream->readFixSize(mask, sizeof(mask)) <= 0)
240 {
241 break;
242 }
243 }
244 data.resize(cur_len + length);
245 if (stream->readFixSize(&data[cur_len], length) <= 0)
246 {
247 break;
248 }
249 if (ws_head.mask)
250 {
251 for (int i = 0; i < static_cast<int>(length); ++i)
252 {
253 data[cur_len + i] ^= mask[i % 4];
254 }
255 }
256 cur_len += length;
257
258 if (!opcode && ws_head.opcode != WSFrameHead::CONTINUE)
259 {
260 opcode = ws_head.opcode;
261 }
262
263 if (ws_head.fin)
264 {
265 LOG_DEBUG(g_logger) << data;
266 return std::make_shared<WSFrameMessage>(opcode, std::move(data));
267 }
268 }
269 else
270 {
271 LOG_DEBUG(g_logger)
272 << "Invalid WebSocket opcode received. "
273 << "Opcode: " << ws_head.opcode;
274 }
275 }
276 while (true);
277 stream->close();
278 return nullptr;
279 }

281 int32_t WSSendMessage(stream::Stream* stream, const std::shared_ptr<WSFrameMessage>& msg, const bool client,
282 const bool fin)
283 {
284 do
285 {
286 WSFrameHead ws_head = {};
287 ws_head.fin = fin;
288 ws_head.opcode = msg->getOpcode();
289 ws_head.mask = client;
290 const uint64_t size = msg->getData().size();
291 if (size < 126)
292 {
293 ws_head.payload = size;
294 }
295 else if (size < 65536)
296 {
297 ws_head.payload = 126;
298 }
299 else
300 {
301 ws_head.payload = 127;
302 }
303
304 if (stream->writeFixSize(&ws_head, sizeof(ws_head)) <= 0)
305 {
306 break;
307 }
308 if (ws_head.payload == 126)
309 {
310 uint16_t len = size;
311 len = base::byteswapOnLittleEndian(len);
312 if (stream->writeFixSize(&len, sizeof(len)) <= 0)
313 {
314 break;
315 }
316 }
317 else if (ws_head.payload == 127)
318 {
319 uint64_t len = base::byteswapOnLittleEndian(size);
320 if (stream->writeFixSize(&len, sizeof(len)) <= 0)
321 {
322 break;
323 }
324 }
325 if (client)
326 {
327 char mask[4];
328 const uint32_t rand_value = rand();
329 memcpy(mask, &rand_value, sizeof(mask));
330 std::string& data = msg->getData();
331 for (size_t i = 0; i < data.size(); ++i)
332 {
333 data[i] ^= mask[i % 4];
334 }
335
336 if (stream->writeFixSize(mask, sizeof(mask)) <= 0)
337 {
338 break;
339 }
340 }
341 if (stream->writeFixSize(msg->getData().c_str(), size) <= 0)
342 {
343 break;
344 }
345 return size + sizeof(ws_head);
346 }
347 while (false);
348 stream->close();
349 return -1;
350 }
