99 {
100 std::string result;
101 result.reserve(str.size()); // 预分配内存，减少多次分配
102
103 const char* end = str.c_str() + str.length();
104 for (const char* c = str.c_str(); c < end; ++c)
105 {
106 if (*c == '+' && space_as_plus)
107 {
108 result.push_back(' '); // 将 "+" 转换为空格

117 c += 2; // 跳过已处理的两个字符
118 }
119 else

284 in_addr_t GetIPv4Inet()
285 {
286 ifaddrs* ifas = nullptr;
287 const in_addr_t localhost = inet_addr("127.0.0.1");
288
289 // 获取网络接口列表
290 if (getifaddrs(&ifas) != 0)
291 {
292 LOG_ERROR(g_logger)
293 << "getifaddrs failed. "
294 << "Error number: " << errno
295 << " | Error string: " << strerror(errno);
296 return localhost; // 出错时返回回环地址
297 }
298
299 in_addr_t ipv4 = localhost; // 默认设置为回环地址
300
301 // 遍历所有接口地址
302 for (const ifaddrs* ifa = ifas; ifa != nullptr; ifa = ifa->ifa_next)
303 {
304 // 只处理IPv4地址
305 if (ifa->ifa_addr && ifa->ifa_addr->sa_family == AF_INET)
306 {
307 // 排除环回接口
308 if (strncasecmp(ifa->ifa_name, "lo", 2) != 0)
309 {
310 ipv4 = reinterpret_cast<sockaddr_in*>(ifa->ifa_addr)->sin_addr.s_addr;
311 // 如果发现有效IPv4地址且不为回环地址，更新 ipv4
312 if (ipv4 != localhost)
313 {
314 break; // 找到有效的 IPv4 地址，提前返回
315 }
316 }
317 }
318 }
319 freeifaddrs(ifas);
320 return ipv4;
321 }
