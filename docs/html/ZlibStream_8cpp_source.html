194 int ZlibStream::encode(const iovec* value, const uint64_t& size, const bool finish)
195 {
196 int ret = 0;
197 int flush = 0;
198 for (uint64_t i = 0; i < size; ++i)
199 {
200 m_zstream.avail_in = value[i].iov_len;
201 m_zstream.next_in = static_cast<Bytef*>(value[i].iov_base);
202
203 flush = finish ? (i == size - 1 ? Z_FINISH : Z_NO_FLUSH) : Z_NO_FLUSH;
204
205 iovec* ivc = nullptr;
206 do
207 {
208 if (!m_buffs.empty() && m_buffs.back().iov_len != m_buffSize)
209 {
210 ivc = &m_buffs.back();
211 }
212 else
213 {
214 iovec vc = {};
215 vc.iov_base = malloc(m_buffSize);
216 vc.iov_len = 0;
217 m_buffs.push_back(vc);
218 ivc = &m_buffs.back();
219 }
220
221 m_zstream.avail_out = m_buffSize - ivc->iov_len;
222 m_zstream.next_out = static_cast<Bytef*>(ivc->iov_base) + ivc->iov_len;
223
224 ret = deflate(&m_zstream, flush);
225 if (ret == Z_STREAM_ERROR)
226 {
227 return ret;
228 }
229 ivc->iov_len = m_buffSize - m_zstream.avail_out;
230 }
231 while (m_zstream.avail_out == 0);
232 }
233 if (flush == Z_FINISH)
234 {
235 deflateEnd(&m_zstream);
236 }
237 return Z_OK;
