41 const auto parser = std::make_shared<HttpResponseParser>();
42 const uint64_t buff_size = HttpRequestParser::GetHttpRequestBufferSize();
43 std::vector<char> buffer(buff_size);
44 char* data = buffer.data();
45 int offset = 0;
46 do
47 {
48 auto len = read(data + offset, buff_size - offset);
49 if (len <= 0)
50 {
51 close();
52 return nullptr;
53 }
54 len += offset;
55 data[len] = '\0';
56 const size_t nparse = parser->execute(data, len, false);
57 if (parser->hasError())
58 {
59 close();
60 return nullptr;
61 }
62 offset = len - nparse;
63 if (offset == static_cast<int>(buff_size))
64 {
65 close();
66 return nullptr;
67 }
68 if (parser->isFinished())
69 {
70 break;
71 }
72 }
73 while (true);
74 auto& client_parser = parser->getParser();
75 std::string body;
76 if (client_parser.chunked)
77 {
78 int len = offset;
79 do
80 {
81 bool begin = true;
82 do
83 {
84 if (!begin || len == 0)
85 {
86 const auto rt = read(data + len, buff_size - len);
87 if (rt <= 0)
88 {
89 close();
90 return nullptr;
91 }
92 len += rt;
93 }
94 data[len] = '\0';
95 const size_t nparse = parser->execute(data, len, true);
96 if (parser->hasError())
97 {
98 close();
99 return nullptr;
100 }
101 len -= nparse;
102 if (len == static_cast<int>(buff_size))
103 {
104 close();
105 return nullptr;
106 }
107 begin = false;
108 }
109 while (!parser->isFinished());
110
111 if (client_parser.content_len + 2 <= len)
112 {
113 body.append(data, client_parser.content_len);
114 memmove(data, data + client_parser.content_len + 2
115 , len - client_parser.content_len - 2);
116 len -= client_parser.content_len + 2;
117 }
118 else
119 {
120 body.append(data, len);
121 int left = client_parser.content_len - len + 2;
122 while (left > 0)
123 {
124 const auto rt = read(data, left > static_cast<int>(buff_size)
125 ? static_cast<int>(buff_size)
126 : left);
127 if (rt <= 0)
128 {
129 close();
130 return nullptr;
131 }
132 body.append(data, rt);
133 left -= rt;
134 }
135 body.resize(body.size() - 2);
136 len = 0;
137 }
138 }
139 while (!client_parser.chunks_done);
140 }
141 else
142 {
143 if (auto length = parser->getContentLength(); length > 0)
144 {
145 body.resize(length);
146
147 uint64_t len = 0;
148 if (length >= static_cast<size_t>(offset))
149 {
150 memcpy(&body[0], data, offset);
151 len = offset;
152 }
153 else
154 {
155 memcpy(&body[0], data, length);
156 len = length;
157 }
158 length -= offset;
159 if (length > 0)
160 {
161 if (readFixSize(&body[len], length) <= 0)

163 close();
164 return nullptr;
165 }
166 }
167 }

175 if (strcasecmp(content_encoding.c_str(), "gzip") == 0)

178 zs->write(body.c_str(), body.size());
179 zs->flush();
180 zs->getResult().swap(body);
181 }

294 const std::shared_ptr<Uri>& uri,
295 const uint64_t timeout_ms)
296 {
297 const bool is_ssl = uri->getScheme() == "https";
298 const auto address = uri->createAddress();
299 if (!address)
300 {
301 return std::make_shared<HttpResult>(static_cast<int>(HttpResult::Error::INVALID_HOST)
302 , nullptr, "invalid host: " + uri->getHost());
303 }
304 auto sock = is_ssl ? SSLSocket::CreateTCP(address) : Socket::CreateTCP(address);

309 + " errno=" + std::to_string(errno)

312 if (!sock->connect(address, std::chrono::milliseconds::max().count()))

319 const auto rt = conn->sendRequest(req);

323 , nullptr, "send request closed by peer: " + address->toString());

332 if (!rsp)

353 std::shared_ptr<HttpConnection> HttpConnectionPool::getConnection()
354 {
355 const uint64_t now_ms = std::chrono::high_resolution_clock::now().time_since_epoch().count();
356 std::vector<HttpConnection*> invalid_conns;
357 HttpConnection* ptr = nullptr;
358 std::unique_lock lock(m_mutex);
359 while (!m_connections.empty())
360 {
361 auto conn = *m_connections.begin();
362 m_connections.pop_front();
363 if (!conn->isConnected())
364 {
365 invalid_conns.push_back(conn);
366 continue;
367 }
368 if (conn->m_createTime + m_maxAliveTime > now_ms)
369 {
370 invalid_conns.push_back(conn);
371 continue;
372 }
373 ptr = conn;
374 break;
375 }
376 lock.unlock();
377 for (const auto i : invalid_conns)
378 {
379 delete i;
380 }
381 m_total -= invalid_conns.size();
382
383 if (!ptr)
384 {
385 const auto address = Address::LookupAnyIPAddress(m_host);
386 if (!address)
387 {
388 LOG_ERROR(g_logger) << "get address failed:" << m_host;
389 return nullptr;
390 }
391 address->setPort(m_port);
392 const auto sock = m_isHttps ? SSLSocket::CreateTCP(address) : Socket::CreateTCP(address);
393 // const auto sock = Socket::CreateTCP(address);
394 if (!sock)
395 {
396 LOG_ERROR(g_logger) << "create socket fail";
397 return nullptr;
398 }
399 if (!sock->connect(address, std::chrono::milliseconds::max().count()))
400 {
401 LOG_ERROR(g_logger) << "sock connect fail";
402 return nullptr;
403 }
404
405 ptr = new HttpConnection(sock);
406 ++m_total;
407 }
408 return {
409 ptr, [this](auto&& PH1) { HttpConnectionPool::ReleasePtr(std::forward<decltype(PH1)>(PH1), this); }
