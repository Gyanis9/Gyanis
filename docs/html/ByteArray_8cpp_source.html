791 void ByteArray::addCapacity(size_t size)
792 {
793 if (size == 0)
794 {
795 return;
796 }
797 const size_t old_cap = getCapacity();
798 if (old_cap >= size)
799 {
800 return;
801 }
802
803 size = size - old_cap;
804 const size_t count = ceil(1.0 * static_cast<double>(size) / static_cast<double>(m_baseSize));
805 Node* tmp = m_root;
806 while (tmp->next)
807 {
808 tmp = tmp->next;
809 }
810
811 Node* first = nullptr;
812 for (size_t i = 0; i < count; ++i)
813 {
814 tmp->next = new Node(m_baseSize);
815 if (first == nullptr)
816 {
817 first = tmp->next;
818 }
819 tmp = tmp->next;
820 m_capacity += m_baseSize;
821 }
822
823 if (old_cap == 0)
824 {
825 m_current = first;
826 }
