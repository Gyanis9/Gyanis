39 ServerInfo parse_server_info(const std::shared_ptr<web::TcpServer>& server)
40 {
41 ServerInfo info = {};
42 std::string raw = server->toString("");
43
44 // 解析类型
45 size_t type_pos = raw.find("Type: ");
46 if (type_pos != std::string::npos)
47 {
48 size_t end = raw.find(" | ", type_pos);
49 info.type = raw.substr(type_pos + 6, end - (type_pos + 6));
50 }
51
52 // 解析SSL状态
53 size_t ssl_pos = raw.find("SSL enabled: ");
54 if (ssl_pos != std::string::npos)
55 {
56 info.ssl = (raw.substr(ssl_pos + 13, 4) == "true");
57 }
58
59 // 解析地址（取第一个socket地址）
60 size_t addr_start = raw.find("[socket]: ");
61 if (addr_start != std::string::npos)
62 {
63 size_t addr_end = raw.find("address", addr_start + 10);
64 info.address = raw.substr(addr_start + 10, addr_end - (addr_start + 10));
65 }
66
67 // 映射协议版本
68 info.protocol_version =
69 (info.type == "http")
70 ? "HTTP/1.1"
71 : (info.type == "https")
72 ? "HTTP/2"
73 : (info.type == "ws")
74 ? "WebSocket"
75 : info.type;
76
77 return info;
78 }
