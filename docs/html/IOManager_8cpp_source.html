126 {

322 void IOManager::idle()
323 {
324 constexpr uint64_t MAX_EVENTS = 1024;
325 std::vector<epoll_event> events(MAX_EVENTS);
326 while (true)
327 {
328 auto next_timeout = std::chrono::milliseconds::zero();
329 if (LIKELY(stopping(next_timeout)))
330 {
331 LOG_INFO(g_logger)
332 << "IOManager::idle - IOManager idle timeout reached. "
333 << "Stopping and exiting. "
334 << " | Status: Normal";
335 break;
336 }
337 int rt = 0;
338 do
339 {
340 static constexpr std::chrono::milliseconds MAX_TIMEOUT{3000};
341 if (next_timeout != std::chrono::milliseconds::max())
342 {
343 next_timeout = next_timeout > MAX_TIMEOUT ? MAX_TIMEOUT : next_timeout;
344 }
345 else
346 {
347 next_timeout = MAX_TIMEOUT;
348 }
349 rt = epoll_wait(m_epFd, events.data(), MAX_EVENTS, static_cast<int>(next_timeout.count()));
350 if (rt < 0 && errno == EINTR)
351 {
352 continue;
353 }
354 break;
355 }
356 while (true);
357 std::vector<std::function<void()>> callbacks;
358 ListExpiredCb(callbacks);
359 if (!callbacks.empty())
360 {
361 schedule(callbacks.begin(), callbacks.end());
362 callbacks.clear();
363 }
364 for (int i = 0; i < rt; ++i)
365 {
366 auto& event = events[i];
367 if (event.data.fd == m_wakeFd)
368 {
369 uint64_t value = 0;
370 while (eventfd_read(m_wakeFd, &value) > 0)
371 {
372 }
373 continue;
374 }
375 auto* fd_context = static_cast<FdContext*>(event.data.ptr);
376 std::scoped_lock lock1(fd_context->mutex);
377 if (event.events & (EPOLLERR | EPOLLHUP))
378 {
379 event.events |= (EPOLLIN | EPOLLOUT) & fd_context->events;
380 }
381 int real_events = NONE;
382 if (event.events & EPOLLIN)
383 {
384 real_events |= READ;
385 }
386 if (event.events & EPOLLOUT)
387 {
388 real_events |= WRITE;
389 }
390 if ((fd_context->events & real_events) == NONE)
391 {
392 continue;
393 }
394 const int left_events = fd_context->events & ~real_events;
395 const int op = left_events ? EPOLL_CTL_MOD : EPOLL_CTL_DEL;
396 event.events = EPOLLET | left_events;
397 if (epoll_ctl(m_epFd, op, fd_context->fd, &event))
398 {
399 LOG_ERROR(g_logger)
400 << "IOManager::idle - epoll_ctl operation failed. "
401 << "An error occurred while managing the IO events. "
402 << " | Status: Invalid";
403 continue;
404 }
405 if (real_events & READ)
406 {
407 fd_context->triggerEvent(READ);
408 --m_pendingEventCount;
409 }
410 if (real_events & WRITE)
411 {
412 fd_context->triggerEvent(WRITE);
413 --m_pendingEventCount;
414 }
415 }
416 Fiber::Yield();
417 }
