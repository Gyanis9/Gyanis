135 void ModuleManager::del(const std::string& name)
136 {
137 std::unique_lock lock(m_mutex);
138 const auto it = m_modules.find(name);
139 if (it == m_modules.end())
140 {
141 return;
142 }
143 const std::shared_ptr<Module> module = it->second;
144 m_modules.erase(it);
145 m_type2Modules[module->getType()].erase(module->getId());
146 if (m_type2Modules[module->getType()].empty())
147 {
148 m_type2Modules.erase(module->getType());
149 }
150 lock.unlock();
151 module->onUnload();
152 }
